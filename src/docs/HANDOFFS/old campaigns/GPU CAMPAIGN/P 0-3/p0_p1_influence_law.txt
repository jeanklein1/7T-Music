COMMIT [P1a] — the profile struct + the response body
1. CLASS B — beside the contact const cluster, add:

   // ═══ THE INFLUENCE LAW (CONTACT_5) ══════════════════════════════
   // One body, many callers. The PROFILE selects the shape; the law
   // is written once. See the authority table in the P0 banner.
   struct InfluenceProfile {
       radius:        f32,   // shell radius (see shell_vwindow)
       vwindow:       f32,   // <= 0 : spherical gate (3D distance)
                             //  > 0 : CYLINDRICAL gate — planar radius,
                             //        |dy| < vwindow (for hovering bodies)
       presence_gain: f32,   // overlap term (r-d)·gain, dt-scaled, 0 = off
       approach_gain: f32,   // approach term v_ap·gain, NOT dt-scaled, 0 = off
       falloff_mix:   f32,   // 0 = flat across the shell, 1 = linear (1-d/r)
       cap:           f32,   // max magnitude of the summed response
       yield_share:   f32,   // 0..1 — how much of it THIS body takes
   }

   // Returns the PLANAR response for `self`, in wu/s (the caller adds
   // it to velocity, and — where the caller integrates inline — to
   // position ×dt, per K1b).
   fn influence_response(self_pos: vec3<f32>, self_vel: vec2<f32>,
                         other_pos: vec3<f32>, other_vel: vec2<f32>,
                         p: InfluenceProfile, dt: f32) -> vec2<f32> {
       let d3 = self_pos - other_pos;
       let d_pl = sqrt(max(d3.x*d3.x + d3.z*d3.z, 0.0001));
       // THE GATE — spherical or cylindrical (hovering bodies get a
       // column, not a ball: you shove a floating cube by walking
       // under it).
       var d_gate = d_pl;
       if (p.vwindow <= 0.0) {
           d_gate = sqrt(max(dot(d3, d3), 0.0001));
       } else if (abs(d3.y) > p.vwindow) {
           return vec2(0.0);
       }
       if (d_gate >= p.radius) { return vec2(0.0); }
       let dir = vec2(d3.x, d3.z) / d_pl;              // other -> self
       let fall = mix(1.0, clamp(1.0 - d_gate / p.radius, 0.0, 1.0),
                      clamp(p.falloff_mix, 0.0, 1.0));
       // PRESENCE — occupancy. Impulse: dt-scaled (K1 law).
       var mag = (p.radius - d_gate) * p.presence_gain * dt;
       // APPROACH — motion. Velocity floor: NOT dt-scaled (K1 law).
       var esc = dir;
       if (p.approach_gain > 0.0) {
           let v_ap = max(0.0, dot(other_vel, dir));
           if (v_ap > 0.001) {
               let deficit = v_ap * p.approach_gain * fall
                           - dot(self_vel, dir);
               if (deficit > 0.0) {
                   mag += deficit;
                   let tang = vec2(-dir.y, dir.x)
                            * sign(other_vel.x*dir.y - other_vel.y*dir.x + 0.000001);
                   esc = normalize(dir + tang * 0.6);   // the matador split
               }
           }
       }
       return esc * min(mag * fall, p.cap) * p.yield_share;
   }
